cmake_minimum_required(VERSION 3.18)
project(conversion_inference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted ONNX Runtime C/C++ SDK (include/ and lib/)")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
    HINTS "${ONNXRUNTIME_ROOT}/include" PATH_SUFFIXES onnxruntime/core/session)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib")
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
    message(FATAL_ERROR "ONNX Runtime SDK not found. Pass -DONNXRUNTIME_ROOT=/path/to/sdk with include/ and lib/.")
endif()

add_executable(predict src/main.cpp src/model.cpp src/benchmark.cpp)
target_include_directories(predict PRIVATE include)
target_include_directories(predict SYSTEM PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(predict PRIVATE "${ONNXRUNTIME_LIBRARY}")
get_filename_component(ONNXRUNTIME_LIB_DIR "${ONNXRUNTIME_LIBRARY}" DIRECTORY)
set_target_properties(predict PROPERTIES BUILD_RPATH "${ONNXRUNTIME_LIB_DIR}")
if(MSVC)
    target_compile_options(predict PRIVATE /W4)
else()
    target_compile_options(predict PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_custom_command(TARGET predict POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:predict>")
endif()
